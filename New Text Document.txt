function escapeHTML(value){

    if(value === null || value === undefined){
        return "";
    }

    return String(value)
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
}

async function adminLogin(){

    const email =
        document.getElementById("email")
        .value
        .trim();

    const password =
        document.getElementById("password")
        .value;

    document
        .getElementById("loginError")
        .classList.add("hidden");


    if(!email || !password){

        showError("Please enter email and password.");

        return;
    }


    if(email.toLowerCase() !== ADMIN_EMAIL.toLowerCase()){

        showError("This account is not authorized as MIC POWER Admin.");

        return;
    }


    const {
        data,
        error
    } =
    await supabaseClient.auth.signInWithPassword({

        email:email,
        password:password

    });


    if(error){

        showError(error.message);

        return;
    }


    if(
        !data ||
        !data.user ||
        data.user.email.toLowerCase()
        !== ADMIN_EMAIL.toLowerCase()
    ){

        await supabaseClient.auth.signOut();

        showError("Admin authorization failed.");

        return;
    }


    showApp();

}

/* ================================
   STATISTICS
================================ */

async function loadStatistics(){

    const {
        data,
        error
    } =
    await supabaseClient
        .from("releases")
        .select("id,status");


    if(error){

        console.error(error);

        return;
    }


    const releases = data || [];


    const total =
        releases.length;

    const submitted =
        releases.filter(
            r => r.status === "Submitted"
        ).length;

    const approved =
        releases.filter(
            r => r.status === "Approved"
        ).length;

    const rejected =
        releases.filter(
            r => r.status === "Rejected"
        ).length;


    document.getElementById("totalCount")
        .textContent = total;

    document.getElementById("submittedCount")
        .textContent = submitted;

    document.getElementById("approvedCount")
        .textContent = approved;

    document.getElementById("rejectedCount")
        .textContent = rejected;
}

/* ================================
   LOAD RELEASES
================================ */

async function loadReleases(statusFilter = "Submitted"){

    const list =
        document.getElementById("releaseList");

    list.innerHTML =
        "<div class='empty'>Loading releases...</div>";


    let query =
        supabaseClient
            .from("releases")
            .select("*")
            .order("created_at",{ascending:false});


    if(statusFilter !== "ALL"){

        query =
            query.eq(
                "status",
                statusFilter
            );

    }


    const {
        data,
        error
    } =
    await query;


    if(error){

        console.error(error);

        list.innerHTML =
            `<div class="error">
                ${escapeHTML(error.message)}
            </div>`;

        return;
    }


    if(!data || data.length === 0){

        list.innerHTML =
            `<div class="empty">
                No ${escapeHTML(statusFilter)}
                releases found.
            </div>`;

        return;
    }


    list.innerHTML = "";


    data.forEach(release => {

        const card =
            document.createElement("div");

        card.className = "release";


        const title =
            release.song_title ||
            release.title ||
            "Untitled";


        const artist =
            release.artist_name ||
            "Unknown Artist";


        const cover =
            release.cover_url ||
            release.cover_art_url ||
            "";


        const audio =
            release.audio_url ||
            "";


        const status =
            release.status ||
            "Draft";


        let actionButtons = "";


        if(status === "Submitted"){

            actionButtons = `

                <button
                    class="approve"
                    onclick="approveRelease('${release.id}')">

                    ✅ Approve

                </button>

                <button
                    class="reject"
                    onclick="rejectRelease('${release.id}')">

                    ❌ Reject

                </button>

            `;

        }


        card.innerHTML = `

            <div class="release-top">

                ${
                    cover
                    ?
                    `<img
                        class="cover"
                        src="${escapeHTML(cover)}"
                        alt="Cover">`
                    :
                    `<div class="cover"></div>`
                }


                <div class="release-info">

                    <div class="release-title">
                        ${escapeHTML(title)}
                    </div>

                    <div class="artist">
                        ${escapeHTML(artist)}
                    </div>


                    <div class="details">

                        <div>
                            <strong>Type:</strong>
                            ${escapeHTML(release.release_type || "-")}
                        </div>

                        <div>
                            <strong>Genre:</strong>
                            ${escapeHTML(release.genre || "-")}
                        </div>

                        <div>
                            <strong>Language:</strong>
                            ${escapeHTML(release.language || "-")}
                        </div>

                        <div>
                            <strong>Release Date:</strong>
                            ${escapeHTML(release.release_date || "-")}
                        </div>

                    </div>


                    <span class="status ${status.toLowerCase()}">
                        ${escapeHTML(status)}
                    </span>

                </div>

            </div>


            ${
                audio
                ?
                `<audio
                    controls
                    preload="none"
                    src="${escapeHTML(audio)}">
                </audio>`
                :
                `<div class="error">
                    Audio file URL is missing.
                </div>`
            }


            ${
                actionButtons
                ?
                `<div class="actions">
                    ${actionButtons}
                </div>`
                :
                ""
            }

        `;


        list.appendChild(card);

    });

}
